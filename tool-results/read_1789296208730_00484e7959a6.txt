   559→   559→function QRPlaceholder() {
   560→   560→  // Deterministic pseudo-QR grid — 21x21 modules
   561→   561→  const cells = useMemo(() => {
   562→   562→    const N = 21;
   563→   563→    const grid: boolean[][] = Array.from({ length: N }, () =>
   564→   564→      Array.from({ length: N }, () => false),
   565→   565→    );
   566→   566→    // Finder patterns (top-left, top-right, bottom-left)
   567→   567→    const stamp = (r0: number, c0: number) => {
   568→   568→      for (let r = 0; r < 7; r++) {
   569→   569→        for (let c = 0; c < 7; c++) {
   570→   570→          const onOuter = r === 0 || r === 6 || c === 0 || c === 6;
   571→   571→          const onInner =
   572→   572→            r >= 2 && r <= 4 && c >= 2 && c <= 4 && (r === 2 || r === 4 || c === 2 || c === 4 || (r === 3 && c === 3));
   573→   573→          grid[r0 + r][c0 + c] = onOuter || (r >= 2 && r <= 4 && c >= 2 && c <= 4 && r === 3 && c === 3) || onInner;
   574→   574→        }
   575→   575→      }
   576→   576→    };
   577→   577→    stamp(0, 0);
   578→   578→    stamp(0, N - 7);
   579→   579→    stamp(N - 7, 0);
   580→   580→    // Pseudo-random fill
   581→   581→    let seed = 7;
   582→   582→    const rand = () => {
   583→   583→      seed = (seed * 9301 + 49297) % 233280;
   584→   584→      return seed / 233280;
   585→   585→    };
   586→   586→    for (let r = 0; r < N; r++) {
   587→   587→      for (let c = 0; c < N; c++) {
   588→   588→        // Skip finder zones
   589→   589→        const inFinder =
   590→   590→          (r < 8 && c < 8) ||
   591→   591→          (r < 8 && c > N - 9) ||
   592→   592→          (r > N - 9 && c < 8);
   593→   593→        if (inFinder) continue;
   594→   594→        grid[r][c] = rand() > 0.55;
   595→   595→      }
   596→   596→    }
   597→   597→    return grid;
   598→   598→  }, []);
   599→   599→
   600→   600→  return (
   601→   601→    <div className="grid h-36 w-36 grid-cols-[repeat(21,1fr)] gap-0 overflow-hidden rounded-md">
   602→   602→      {cells.flatMap((row, ri) =>
   603→   603→        row.map((on, ci) => (
   604→   604→          <div
   605→   605→            key={`${ri}-${ci}`}
   606→   606→            className={on ? "bg-slate-900" : "bg-white"}
   607→   607→            style={{ aspectRatio: "1 / 1" }}
   608→   608→          />
   609→   609→        )),
   610→   610→      )}
   611→   611→    </div>
   612→   612→  );
   613→   613→}
