   499→

   502→      className={cn(

   506→      )}

   509→        {row.rank}
   510→      </TableCell>
   511→      <TableCell className="py-3">
   512→        <div className="flex items-center gap-2.5">
   513→          <div className="grid h-8 w-8 shrink-0 place-items-center rounded-full bg-violet-500/10 text-xs font-semibold text-violet-700 dark:text-violet-300">

   517→            <p className="truncate text-sm font-medium">{row.name}</p>
   518→            <p className="truncate text-xs text-muted-foreground">
   519→              <span className="font-mono">{row.id}</span>
   520→              {row.offerSent && (
   521→                <span className="ml-2 inline-flex items-center gap-1 rounded-full border border-emerald-400/40 bg-emerald-500/10 px-1.5 py-0.5 text-[10px] font-medium text-emerald-700 dark:text-emerald-300">

   523→                </span>
   524→              )}
   525→            </p>
   526→          </div>
   527→        </div>

   529→
   530→      {/* Rounds mini-breakdown */}
   531→      <TableCell className="hidden md:table-cell">
   532→        <div className="flex flex-wrap gap-1.5">
   533→          {row.rounds.map((r) => (
   534→            <div
   535→              key={r.round}
   536→              className="flex items-center gap-1.5 rounded-md border border-black/5 bg-muted/30 px-2 py-1 text-[10px] dark:border-white/5"
   537→            >
   538→              {r.round === "Aptitude" ? (
   539→                <Brain className="h-3 w-3 text-violet-500 dark:text-violet-300" />
   540→              ) : r.round === "Coding" ? (
