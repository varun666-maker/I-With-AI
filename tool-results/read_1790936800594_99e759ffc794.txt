   949→   949→function QRPlaceholder() {
   950→   950→  // Deterministic pseudo-QR grid. 21x21 modules
   951→   951→  const cells = useMemo(() => {
   952→   952→    const N = 21;
   953→   953→    const grid: boolean[][] = Array.from({ length: N }, () =>
   954→   954→      Array.from({ length: N }, () => false),
   955→   955→    );
   956→   956→    const stamp = (r0: number, c0: number) => {
   957→   957→      for (let r = 0; r < 7; r++) {
   958→   958→        for (let c = 0; c < 7; c++) {
   959→   959→          const onOuter = r === 0 || r === 6 || c === 0 || c === 6;
   960→   960→          const onInner =
   961→   961→            r >= 2 && r <= 4 && c >= 2 && c <= 4 && (r === 2 || r === 4 || c === 2 || c === 4 || (r === 3 && c === 3));
   962→   962→          grid[r0 + r][c0 + c] = onOuter || (r >= 2 && r <= 4 && c >= 2 && c <= 4 && r === 3 && c === 3) || onInner;
   963→   963→        }
   964→   964→      }
   965→   965→    };
   966→   966→    stamp(0, 0);
   967→   967→    stamp(0, N - 7);
   968→   968→    stamp(N - 7, 0);
   969→   969→    let seed = 7;
   970→   970→    const rand = () => {
   971→   971→      seed = (seed * 9301 + 49297) % 233280;
   972→   972→      return seed / 233280;
   973→   973→    };
   974→   974→    for (let r = 0; r < N; r++) {
   975→   975→      for (let c = 0; c < N; c++) {
   976→   976→        const inFinder =
   977→   977→          (r < 8 && c < 8) ||
   978→   978→          (r < 8 && c > N - 9) ||
   979→   979→          (r > N - 9 && c < 8);
   980→   980→        if (inFinder) continue;
   981→   981→        grid[r][c] = rand() > 0.55;
   982→   982→      }
   983→   983→    }
   984→   984→    return grid;
   985→   985→  }, []);
   986→   986→
   987→   987→  return (
   988→   988→    <div className="grid h-36 w-36 grid-cols-[repeat(21,1fr)] gap-0 overflow-hidden rounded-md">
   989→   989→      {cells.flatMap((row, ri) =>
   990→   990→        row.map((on, ci) => (
   991→   991→          <div
   992→   992→            key={`${ri}-${ci}`}
   993→   993→            className={on ? "bg-slate-900" : "bg-white"}
   994→   994→            style={{ aspectRatio: "1 / 1" }}
   995→   995→          />
   996→   996→        )),
   997→   997→      )}
   998→   998→    </div>
   999→   999→  );
  1000→  1000→}
