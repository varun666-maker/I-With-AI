   452→
   453→function SystemCheckCard({
   454→  icon: Icon,
   455→  label,
   456→  status,
   457→  onTest,
   458→}: {
   459→  icon: React.ComponentType<{ className?: string }>;
   460→  label: string;
   461→  status: "idle" | "ok" | "fail";
   462→  onTest: () => void;
   463→}) {
   464→  return (
   465→    <div className="rounded-xl border border-white/10 bg-white/[0.03] p-4">
   466→      <div className="flex items-center justify-between">
   467→        <div className="flex items-center gap-3">
   468→          <span className="grid h-10 w-10 place-items-center rounded-lg bg-white/5 text-muted-foreground ring-1 ring-inset ring-white/10">
   469→            <Icon className="h-5 w-5" />
   470→          </span>
   471→          <div>
   472→            <p className="text-sm font-medium">{label}</p>
   473→            <p className="text-[11px] text-muted-foreground">
   474→              {status === "idle" && "Not tested yet"}
   475→              {status === "ok" && "Working ✓"}
   476→              {status === "fail" && "Not detected"}
   477→            </p>
   478→          </div>
