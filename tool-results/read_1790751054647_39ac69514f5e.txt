   549→function QRPlaceholder() {
   550→  // Deterministic pseudo-QR grid. 21x21 modules
   551→  const cells = useMemo(() => {
   552→    const N = 21;
   553→    const grid: boolean[][] = Array.from({ length: N }, () =>
   554→      Array.from({ length: N }, () => false),
   555→    );
   556→    // Finder patterns (top-left, top-right, bottom-left)
   557→    const stamp = (r0: number, c0: number) => {
   558→      for (let r = 0; r < 7; r++) {
   559→        for (let c = 0; c < 7; c++) {
   560→          const onOuter = r === 0 || r === 6 || c === 0 || c === 6;
   561→          const onInner =
   562→            r >= 2 && r <= 4 && c >= 2 && c <= 4 && (r === 2 || r === 4 || c === 2 || c === 4 || (r === 3 && c === 3));
   563→          grid[r0 + r][c0 + c] = onOuter || (r >= 2 && r <= 4 && c >= 2 && c <= 4 && r === 3 && c === 3) || onInner;
   564→        }
   565→      }
   566→    };
   567→    stamp(0, 0);
   568→    stamp(0, N - 7);
   569→    stamp(N - 7, 0);
   570→    // Pseudo-random fill
   571→    let seed = 7;
   572→    const rand = () => {
   573→      seed = (seed * 9301 + 49297) % 233280;
   574→      return seed / 233280;
   575→    };
   576→    for (let r = 0; r < N; r++) {
   577→      for (let c = 0; c < N; c++) {
   578→        // Skip finder zones
   579→        const inFinder =
   580→          (r < 8 && c < 8) ||
   581→          (r < 8 && c > N - 9) ||
   582→          (r > N - 9 && c < 8);
   583→        if (inFinder) continue;
   584→        grid[r][c] = rand() > 0.55;
   585→      }
   586→    }
   587→    return grid;
   588→  }, []);
   589→
   590→  return (
   591→    <div className="grid h-36 w-36 grid-cols-[repeat(21,1fr)] gap-0 overflow-hidden rounded-md">
   592→      {cells.flatMap((row, ri) =>
   593→        row.map((on, ci) => (
   594→          <div
   595→            key={`${ri}-${ci}`}
   596→            className={on ? "bg-slate-900" : "bg-white"}
   597→            style={{ aspectRatio: "1 / 1" }}
   598→          />
   599→        )),
   600→      )}
   601→    </div>
   602→  );
   603→}
