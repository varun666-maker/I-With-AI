   734→
   735→function LoginPanel({ onLogin }: { onLogin: () => void }) {
   736→  const [email, setEmail] = useState("");
   737→  const [pwd, setPwd] = useState("");
   738→  const [loading, setLoading] = useState(false);
   739→  const [forgot, setForgot] = useState(false);
   740→  const [resetSent, setResetSent] = useState(false);
   741→  const [showPwd, setShowPwd] = useState(false);
   742→  const [errors, setErrors] = useState<{ email?: string; pwd?: string }>({});
   743→  const [touched, setTouched] = useState<{ email?: boolean; pwd?: boolean }>({});
   744→
   745→  const validateEmail = (v: string) => {
   746→    if (!v) return "Email is required";
   747→    if (!/^[^\s@]+@[^\s@]+\.[^\s@]+$/.test(v)) return "Enter a valid email address";
   748→    return undefined;
   749→  };
   750→  const validatePwd = (v: string) => {
   751→    if (!v) return "Password is required";
   752→    if (v.length < 8) return "Password must be at least 8 characters";
   753→    return undefined;
   754→  };
   755→
   756→  const valid = !validateEmail(email) && !validatePwd(pwd);
   757→
   758→  const login = () => {
   759→    const newErrors = { email: validateEmail(email), pwd: validatePwd(pwd) };
   760→    setErrors(newErrors);
   761→    setTouched({ email: true, pwd: true });
   762→    if (newErrors.email || newErrors.pwd) return;
   763→    setLoading(true);
   764→    setTimeout(() => {
   765→      setLoading(false);
   766→      onLogin();
   767→    }, 900);
   768→  };
   769→
   770→  return (
   771→    <div>
   772→      <h3 className="text-lg font-semibold">Welcome back</h3>
   773→      <p className="mt-1 text-sm text-muted-foreground">Log in to continue your prep journey.</p>
   774→
   775→      {!forgot ? (
   776→        <>
   777→          <div className="mt-5 space-y-3">
   778→            <div>
   779→              <Label className="text-xs text-muted-foreground">Email Address</Label>
   780→              <div className="relative mt-1">
   781→                <Mail className="absolute left-3 top-1/2 h-4 w-4 -translate-y-1/2 text-muted-foreground" />
   782→                <Input
   783→                  type="email"
   784→                  value={email}
   785→                  onChange={(e) => {
   786→                    setEmail(e.target.value);
   787→                    if (touched.email) setErrors((p) => ({ ...p, email: validateEmail(e.target.value) }));
   788→                  }}
   789→                  onBlur={() => {
   790→                    setTouched((p) => ({ ...p, email: true }));
   791→                    setErrors((p) => ({ ...p, email: validateEmail(email) }));
   792→                  }}
   793→                  placeholder="you@email.com"
   794→                  aria-invalid={!!errors.email && touched.email}
   795→                  aria-describedby={errors.email && touched.email ? "login-email-error" : undefined}
   796→                  className={cn(
   797→                    "rounded-lg bg-white/5 pl-9 border-white/10",
   798→                    errors.email && touched.email && "border-rose-400/60 focus-visible:ring-rose-400/30"
   799→                  )}
   800→                />
   801→              </div>
   802→              {errors.email && touched.email && (
   803→                <p id="login-email-error" className="mt-1 text-xs text-rose-500 dark:text-rose-400" role="alert">
   804→                  {errors.email}
   805→                </p>
   806→              )}
   807→            </div>
   808→            <div>
   809→              <Label className="text-xs text-muted-foreground">Password</Label>
   810→              <div className="relative mt-1">
   811→                <Lock className="absolute left-3 top-1/2 h-4 w-4 -translate-y-1/2 text-muted-foreground" />
   812→                <Input
   813→                  type={showPwd ? "text" : "password"}
   814→                  value={pwd}
   815→                  onChange={(e) => {
   816→                    setPwd(e.target.value);
   817→                    if (touched.pwd) setErrors((p) => ({ ...p, pwd: validatePwd(e.target.value) }));
   818→                  }}
   819→                  onBlur={() => {
   820→                    setTouched((p) => ({ ...p, pwd: true }));
   821→                    setErrors((p) => ({ ...p, pwd: validatePwd(pwd) }));
   822→                  }}
   823→                  placeholder="••••••••"
   824→                  aria-invalid={!!errors.pwd && touched.pwd}
   825→                  aria-describedby={errors.pwd && touched.pwd ? "login-pwd-error" : undefined}
   826→                  className={cn(
   827→                    "rounded-lg bg-white/5 pl-9 pr-9 border-white/10",
   828→                    errors.pwd && touched.pwd && "border-rose-400/60 focus-visible:ring-rose-400/30"
   829→                  )}
   830→                />
   831→                <button
   832→                  type="button"
   833→                  onClick={() => setShowPwd((s) => !s)}
   834→                  className="absolute right-2 top-1/2 -translate-y-1/2 rounded-md p-1 text-muted-foreground hover:text-foreground"
   835→                  aria-label={showPwd ? "Hide password" : "Show password"}
   836→                >
   837→                  {showPwd ? <EyeOff className="h-4 w-4" /> : <Eye className="h-4 w-4" />}
   838→                </button>
   839→              </div>
   840→              {errors.pwd && touched.pwd && (
   841→                <p id="login-pwd-error" className="mt-1 text-xs text-rose-500 dark:text-rose-400" role="alert">
   842→                  {errors.pwd}
   843→                </p>
   844→              )}
   845→            </div>
   846→            <div className="flex items-center justify-between">
   847→              <label className="flex items-center gap-2 text-xs text-muted-foreground">

   865→            className="mt-5 w-full rounded-full bg-gradient-to-r from-violet-500 to-fuchsia-500 text-white border-0 hover:from-violet-400 hover:to-fuchsia-400 disabled:opacity-50"
   866→          >
   867→            {loading ? (
   868→              <>
   869→                <Loader2 className="mr-2 h-4 w-4 animate-spin" /> Signing in…
   870→              </>
   871→            ) : (
   872→              <>
   873→                Login <ArrowRight className="ml-2 h-4 w-4" />
   874→              </>
   875→            )}
   876→          </MagneticButton>
   877→        </>
   878→      ) : (
   879→        <AnimatePresence mode="wait">
   880→          {!resetSent ? (
   881→            <motion.div
   882→              key="forgot-form"
   883→              initial={{ opacity: 0, y: 8 }}
   884→              animate={{ opacity: 1, y: 0 }}
   885→              exit={{ opacity: 0, y: -8 }}

   887→            >

   889→                Enter your email and we&apos;ll send a reset link.
   890→              </p>

   892→                <Label className="text-xs text-muted-foreground">Email Address</Label>

   895→                  <Input
