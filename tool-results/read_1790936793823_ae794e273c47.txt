   856→/**
   857→ * Build a CSV template (with header row + 2 example rows) for either
   858→ * aptitude or coding, then trigger a browser download. Pure client-side.
   859→ */
   860→function downloadCsvTemplate(kind: "aptitude" | "coding") {
   861→  if (typeof window === "undefined") return;
   862→  const isApt = kind === "aptitude";
   863→  const header = isApt
   864→    ? [
   865→        "category",
   866→        "sub_topic",
   867→        "difficulty",
   868→        "question_text",
   869→        "option_a",
   870→        "option_b",
   871→        "option_c",
   872→        "option_d",
   873→        "correct",
   874→        "explanation",
   875→      ].join(",")
   876→    : [
   877→        "title",
   878→        "difficulty",
   879→        "type",
   880→        "statement",
   881→        "signatures",
   882→        "visible_tests",
   883→        "hidden_tests",
   884→        "time_limit_ms",
   885→        "memory_limit_kb",
   886→        "schema",
   887→      ].join(",");
   888→
   889→  const esc = (s: string) => `"${s.replace(/"/g, '""')}"`;
   890→  const rows = isApt
   891→    ? [
   892→        [
   893→          "Quant",
   894→          "Time & Work",
   895→          "Medium",
   896→          esc("If 6 men can complete a task in 12 days, how long will 9 men take?"),
   897→          esc("8 days"),
   898→          esc("9 days"),
   899→          esc("6 days"),
   900→          esc("18 days"),
   901→          "A",
   902→          esc("Total man-days = 6 × 12 = 72. With 9 men, time = 72 / 9 = 8 days."),
   903→        ].join(","),
   904→        [
   905→          "Logical",
   906→          "Series",
   907→          "Easy",
   908→          esc("Find the next term: 2, 6, 12, 20, 30, ?"),
   909→          esc("40"),
   910→          esc("42"),
   911→          esc("44"),
   912→          esc("46"),
   913→          "B",
   914→          esc("Differences: 4, 6, 8, 10, 12. Next term = 30 + 12 = 42."),
   915→        ].join(","),
   916→      ]
   917→    : [
   918→        [
   919→          esc("Two Sum (Sorted)"),
   920→          "Easy",
   921→          "DSA",
   922→          esc("Given a sorted array of integers and a target, return the 1-based indices of the two numbers that add to target."),
   923→          esc("function twoSum(arr: number[], target: number): number[]"),
   924→          "3",
   925→          "7",
   926→          "1000",
   927→          "256",
   928→          "",
   929→        ].join(","),
   930→        [
   931→          esc("Top 3 Products by Revenue"),
   932→          "Medium",
   933→          "SQL",
   934→          esc("Given orders(order_id, product_id, qty, price, created_at), return the top 3 products by total revenue in the last 30 days."),
   935→          esc("SELECT product_id, SUM(qty*price) AS revenue ..."),
   936→          "3",
   937→          "7",
   938→          "2000",
   939→          "512",
   940→          esc("CREATE TABLE orders (order_id INT PRIMARY KEY, product_id INT, qty INT, price DECIMAL(10,2), created_at TIMESTAMP);"),
   941→        ].join(","),
   942→      ];
   943→
   944→  const csv = [header, ...rows].join("\n");
   945→  const blob = new Blob([csv], { type: "text/csv;charset=utf-8;" });
   946→  const url = URL.createObjectURL(blob);
   947→  const a = document.createElement("a");
   948→  a.href = url;
   949→  a.download = isApt
   950→    ? "aptitude_questions_template.csv"
   951→    : "coding_problems_template.csv";
   952→  document.body.appendChild(a);
   953→  a.click();
   954→  document.body.removeChild(a);
   955→  setTimeout(() => URL.revokeObjectURL(url), 1000);
   956→}
