    70→    70→ * Prep-mode coding practice. Splits into two views driven by the
    71→    71→ * `codingView` flag in the global nav store:
    72→    72→ *
    73→    73→ *   - "list"   -> LeetCode-style problem index (ProblemListTable) with a

    99→    99→/**
   100→   100→ * Pick a random problem the candidate hasn't solved yet. If everything is
   101→   101→ * solved, fall back to any problem that isn't the one they're currently on.
   102→   102→ */
   103→   103→function pickRandomUnsolved(solvedIds: Set<number>, excludeId?: number): CodingProblem {
   104→   104→  const candidates = CODING_PROBLEMS.filter(
   105→   105→    (p) => !solvedIds.has(p.id) && p.id !== excludeId,
   106→   106→  );
   107→   107→  if (candidates.length > 0) {
   108→   108→    return candidates[Math.floor(Math.random() * candidates.length)];
   109→   109→  }
   110→   110→  const fallback = CODING_PROBLEMS.filter((p) => p.id !== excludeId);
   111→   111→  return fallback[Math.floor(Math.random() * fallback.length)] ?? CODING_PROBLEMS[0];
   112→   112→}

   124→   124→function ProblemListView() {
   125→   125→  const solvedProblems = useNav((s) => s.solvedProblems);
   126→   126→  const attemptedProblems = useNav((s) => s.attemptedProblems);
   127→   127→  const submissions = useNav((s) => s.submissions);
   128→   128→  const setCodingView = useNav((s) => s.setCodingView);
   129→   129→  const setCurrentProblem = useNav((s) => s.setCurrentProblem);
   130→   130→  const setNavigationLocked = useNav((s) => s.setNavigationLocked);
   131→   131→
   132→   132→  // The store persists solved/attempted IDs as strings (localStorage friendly),
   133→   133→  // but ProblemListTable works with numeric IDs. Convert once per render.
   134→   134→  const solvedNums = useMemo(() => solvedProblems.map((id) => Number(id)), [solvedProblems]);
   135→   135→  const attemptedNums = useMemo(
   136→   136→    () => attemptedProblems.map((id) => Number(id)),
   137→   137→    [attemptedProblems],
   138→   138→  );
   139→   139→  const heatSubs = useMemo(
   140→   140→    () => submissions.map((s) => ({ ...s, problemId: Number(s.problemId) })),
   141→   141→    [submissions],
   142→   142→  );
   143→   143→
   144→   144→  const enterProblem = (p: CodingProblem) => {
   145→   145→    setCurrentProblem(String(p.id));
   146→   146→    setCodingView("editor");
   147→   147→    requestFullscreen();
   148→   148→    setNavigationLocked(true);
   149→   149→  };
   150→   150→
   151→   151→  const handleRandom = () => {
   152→   152→    const solved = new Set(solvedNums);
   153→   153→    enterProblem(pickRandomUnsolved(solved));
   154→   154→  };
   155→   155→
   156→   156→  return (
   157→   157→    <div className="min-h-screen bg-background text-foreground">
   158→   158→      <PageWrap wide>
   159→   159→        <BackToHome />
   160→   160→        <SectionHeading
   161→   161→          eyebrow="Coding Arena"
   162→   162→          title={
   163→   163→            <>
   164→   164→              Practice with the{" "}
   165→   165→              <span className="text-violet-500 dark:text-violet-400">AI code reviewer</span>
   166→   166→            </>
   167→   167→          }
   168→   168→          description="A VS Code-like editor with syntax highlighting, 10 test cases per problem, real-time execution, and an AI review that scores complexity, readability, and gives refactor suggestions."
   169→   169→          align="left"
   170→   170→        />
   171→   171→
   172→   172→        <div className="mt-6">
   173→   173→          <ProblemListTable
   174→   174→            solvedProblems={solvedNums}
   175→   175→            attemptedProblems={attemptedNums}
   176→   176→            submissions={heatSubs}
   177→   177→            onPickProblem={enterProblem}
   178→   178→            onRandomQuestion={handleRandom}
   179→   179→          />
   180→   180→        </div>
   181→   181→      </PageWrap>
   182→   182→    </div>
   183→   183→  );
   184→   184→}

   190→   190→function CodingEditorView() {
   191→   191→  const setCodingView = useNav((s) => s.setCodingView);
   192→   192→  const currentProblemId = useNav((s) => s.currentProblemId);
   193→   193→  const solvedProblems = useNav((s) => s.solvedProblems);
   194→   194→  const setCurrentProblem = useNav((s) => s.setCurrentProblem);
   195→   195→  const markSolved = useNav((s) => s.markSolved);
   196→   196→  const markAttempted = useNav((s) => s.markAttempted);
   197→   197→  const recordSubmission = useNav((s) => s.recordSubmission);
   198→   198→  const setNavigationLocked = useNav((s) => s.setNavigationLocked);
   199→   199→  const reduce = useReducedMotion();
   200→   200→
   201→   201→  // Resolve the active problem. Default to the first problem if the store
   202→   202→  // doesn't have one yet (e.g. user deep-linked into the editor view).
   203→   203→  const problem = useMemo<CodingProblem>(() => {
   204→   204→    const id = currentProblemId ? Number(currentProblemId) : null;
   205→   205→    if (id !== null) {
   206→   206→      const found = getProblemById(id);
   207→   207→      if (found) return found;
   208→   208→    }
   209→   209→    return CODING_PROBLEMS[0];
   210→   210→  }, [currentProblemId]);
   211→   211→
   212→   212→  const [language, setLanguage] = useState<string>("Python");
   213→   213→  const [code, setCode] = useState<string>(problem.starterCode.Python ?? "");
   214→   214→  const [customInput, setCustomInput] = useState<string>(
   215→   215→    problem.testCases[0]?.input ?? "",
   216→   216→  );
   217→   217→  const [results, setResults] = useState<TestCaseResult[] | null>(null);
   218→   218→  const [submitted, setSubmitted] = useState(false);
   219→   219→  const [running, setRunning] = useState(false);
   220→   220→  const [confirmLang, setConfirmLang] = useState<string | null>(null);
   221→   221→  const [autoSaved, setAutoSaved] = useState(false);
   222→   222→  const [successOpen, setSuccessOpen] = useState(false);
   223→   223→  const [reenterOpen, setReenterOpen] = useState(false);
   224→   224→  const [fullscreenExits, setFullscreenExits] = useState(0);
   225→   225→  const [tabSwitches] = useState(1);
   226→   226→
   227→   227→  // Reset the editor state whenever the active problem changes (e.g. when the
   228→   228→  // candidate picks "Next Question"). Deferred to a microtask to keep the
   229→   229→  // effect body free of synchronous setState calls.
   230→   230→  useEffect(() => {
   231→   231→    queueMicrotask(() => {
   232→   232→      setLanguage("Python");
   233→   233→      setCode(problem.starterCode.Python ?? "");
   234→   234→      setCustomInput(problem.testCases[0]?.input ?? "");
   235→   235→      setResults(null);
   236→   236→      setSubmitted(false);
   237→   237→    });
   238→   238→  }, [problem]);
   239→   239→
   240→   240→  // Fullscreen lifecycle: request on mount, exit on unmount, and listen for
   241→   241→  // the user pressing Esc so we can show a friendly "re-enter" modal. Prep
   242→   242→  // mode is forgiving, so there are no penalties.
   243→   243→  useEffect(() => {
   244→   244→    requestFullscreen();
   245→   245→    queueMicrotask(() => setNavigationLocked(true));
   246→   246→    const unsubscribe = onFullscreenChange(() => {
   247→   247→      if (!isCurrentlyFullscreen()) {
   248→   248→        queueMicrotask(() => {
   249→   249→          setReenterOpen(true);
   250→   250→          setFullscreenExits((n) => n + 1);
   251→   251→        });
   252→   252→      }
   253→   253→    });
   254→   254→    return () => {
   255→   255→      unsubscribe();
   256→   256→      exitFullscreen();
   257→   257→      queueMicrotask(() => setNavigationLocked(false));
   258→   258→    };
   259→   259→  }, [setNavigationLocked]);
   260→   260→
   261→   261→  // Autosave simulation. The indicator pulses every 30s to reassure the
   262→   262→  // candidate their work is safe.
   263→   263→  useEffect(() => {
   264→   264→    const t = setInterval(() => {
   265→   265→      queueMicrotask(() => setAutoSaved(true));
   266→   266→      setTimeout(() => queueMicrotask(() => setAutoSaved(false)), 1500);
   267→   267→    }, 30000);
   268→   268→    return () => clearInterval(t);
   269→   269→  }, []);
   270→   270→
   271→   271→  const onLangChange = (lang: string) => {
   272→   272→    if (lang === language) return;
   273→   273→    setConfirmLang(lang);
   274→   274→  };
   275→   275→
   276→   276→  const confirmSwitch = () => {
   277→   277→    if (!confirmLang) return;
   278→   278→    setLanguage(confirmLang);
   279→   279→    setCode(problem.starterCode[confirmLang] ?? "");
   280→   280→    setResults(null);
   281→   281→    setSubmitted(false);
   282→   282→    setConfirmLang(null);
   283→   283→  };
   284→   284→
   285→   285→  const run = () => {
   286→   286→    setRunning(true);
   287→   287→    setTimeout(() => {
   288→   288→      setRunning(false);
   289→   289→      // Mock: 8/10 pass. Cases at index 2 and 7 fail so the candidate sees
   290→   290→      // realistic feedback before submitting.
   291→   291→      setResults(
   292→   292→        problem.testCases.map((tc, i) => ({
   293→   293→          id: tc.id,
   294→   294→          status: i === 2 || i === 7 ? ("fail" as const) : ("pass" as const),
   295→   295→          runtime: `${8 + Math.floor(Math.random() * 12)}ms`,
   296→   296→          mem: "14.0MB",
   297→   297→          got: i === 2 ? "wrong answer" : i === 7 ? "timeout" : undefined,
   298→   298→        })),
   299→   299→      );
   300→   300→    }, 900);
   301→   301→  };
   302→   302→
   303→   303→  const submit = () => {
   304→   304→    setRunning(true);
   305→   305→    setTimeout(() => {
   306→   306→      setRunning(false);
   307→   307→      // Mock: all tests pass on submit (prep mode is forgiving). The failure
   308→   308→      // path below stays in place so the success/attempted branching is real.
   309→   309→      const newResults: TestCaseResult[] = problem.testCases.map((tc) => ({
   310→   310→        id: tc.id,
   311→   311→        status: "pass" as const,
   312→   312→        runtime: `${8 + Math.floor(Math.random() * 12)}ms`,
   313→   313→        mem: "14.0MB",
   314→   314→      }));
   315→   315→      setResults(newResults);
   316→   316→      setSubmitted(true);
   317→   317→      const allPassed = newResults.every((r) => r.status === "pass");
   318→   318→      const idStr = String(problem.id);
   319→   319→      if (allPassed) {
   320→   320→        markSolved(idStr);
   321→   321→        recordSubmission(idStr, true);
   322→   322→        queueMicrotask(() => setSuccessOpen(true));
   323→   323→      } else {
   324→   324→        markAttempted(idStr);
   325→   325→        recordSubmission(idStr, false);
   326→   326→      }
   327→   327→    }, 1400);
   328→   328→  };
   329→   329→
   330→   330→  const passCount = results ? results.filter((r) => r.status === "pass").length : 0;
   331→   331→
   332→   332→  const handleBack = () => {
   333→   333→    setCodingView("list");
   334→   334→    exitFullscreen();
   335→   335→    setNavigationLocked(false);
   336→   336→  };
   337→   337→
   338→   338→  const handleNext = () => {
   339→   339→    const solved = new Set(solvedProblems.map((id) => Number(id)));
   340→   340→    const next = pickRandomUnsolved(solved, problem.id);
   341→   341→    setCurrentProblem(String(next.id));
   342→   342→    setSuccessOpen(false);
   343→   343→  };
   344→   344→
   345→   345→  const reenterFullscreen = () => {
   346→   346→    requestFullscreen();
   347→   347→    setReenterOpen(false);
   348→   348→  };
   349→   349→
   350→   350→  const difficultyBadge = DIFFICULTY_BADGE[problem.difficulty];
   351→   351→  const fileExt = language === "Python" ? "py" : "js";
   352→   352→
   353→   353→  return (
   354→   354→    <div className="min-h-screen bg-background text-foreground">
   355→   355→      <PageWrap wide>
   356→   356→        {/* Top bar: Back / Title / Next */}
   357→   357→        <div className="flex flex-col gap-3 rounded-lg border border-black/10 bg-card p-3 shadow-sm sm:flex-row sm:items-center sm:justify-between sm:p-4 dark:border-white/10">
   358→   358→          <Button
   359→   359→            variant="outline"
   360→   360→            size="sm"
   361→   361→            onClick={handleBack}
   362→   362→            className="rounded-full border-black/10 bg-black/[0.02] hover:bg-black/[0.05] dark:border-white/15 dark:bg-white/5 dark:hover:bg-white/10"
   363→   363→          >
   364→   364→            <ArrowLeft className="mr-2 h-4 w-4" /> Back to Problems
   365→   365→          </Button>
   366→   366→          <div className="flex min-w-0 flex-1 items-center justify-center gap-3 px-2 text-center">
   367→   367→            <span className="truncate text-sm font-semibold tracking-tight sm:text-base">
   368→   368→              <span className="font-mono text-muted-foreground">#{problem.id}</span>{" "}
   369→   369→              {problem.title}
   370→   370→            </span>
   371→   371→            <Badge className={cn("shrink-0", difficultyBadge)}>{problem.difficulty}</Badge>
   372→   372→          </div>
   373→   373→          <Button
   374→   374→            size="sm"
   375→   375→            onClick={handleNext}
   376→   376→            className="rounded-full bg-violet-500 text-white hover:bg-violet-600"
   377→   377→          >
   378→   378→            Next Question <ArrowRight className="ml-2 h-4 w-4" />
   379→   379→          </Button>
   380→   380→        </div>
   381→   381→
   382→   382→        <div className="mt-4 grid grid-cols-1 gap-4 lg:grid-cols-2">
   383→   383→          {/* LEFT PANE: problem statement */}

   397→   397→            </div>
