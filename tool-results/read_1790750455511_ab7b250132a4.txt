   115→type ProblemStatus = "unsolved" | "attempted" | "solved";
   116→interface ProblemState {

   126→  got?: string;

   138→  const set = useNav((s) => s.set);
   139→  const setNavigationLocked = useNav((s) => s.setNavigationLocked);
   140→
   141→  const proctor = useProctoring({ mode: "strict" });
   142→  const proctorRef = useRef(proctor);
   143→  useEffect(() => {
   144→    proctorRef.current = proctor;
   145→  });
   146→
   147→  const [language, setLanguage] = useState<string>("Python");
   148→  const [activeIdx, setActiveIdx] = useState(0);
   149→  const [timeLeft, setTimeLeft] = useState(TEST_DURATION_SEC);
   150→  const [submitOpen, setSubmitOpen] = useState(false);
   151→  const [autoSubmitted, setAutoSubmitted] = useState(false);
   152→  const [states, setStates] = useState<Record<number, ProblemState>>(() => {
   153→    const init: Record<number, ProblemState> = {};
   154→    for (const p of PROBLEMS) init[p.id] = emptyState(p);
   155→    return init;
   156→  });
   157→
   158→  // Lock + fullscreen + start monitoring on mount
   159→  useEffect(() => {
   160→    setNavigationLocked(true);
   161→    requestFullscreen();
   162→    proctorRef.current.startMonitoring();
   163→    return () => {
   164→      proctorRef.current.stopMonitoring();
   165→    };
   166→     
   167→  }, []);
   168→
   169→  // Countdown
   170→  useEffect(() => {
   171→    const t = setInterval(() => {
   172→      setTimeLeft((prev) => {
   173→        if (prev <= 1) {
   174→          clearInterval(t);
   175→          queueMicrotask(() => setAutoSubmitted(true));
   176→          return 0;
   177→        }
   178→        return prev - 1;
   179→      });
   180→    }, 1000);
   181→    return () => clearInterval(t);
   182→  }, []);
   183→
   184→  // Auto-submit via proctor
   185→  useEffect(() => {
   186→    if (proctor.isAutoSubmitted) {
   187→      queueMicrotask(() => setAutoSubmitted(true));
   188→    }
   189→  }, [proctor.isAutoSubmitted]);
   190→
   191→  // Strike 3 -> navigate
   192→  useEffect(() => {
   193→    if (proctor.strikes >= 3) {
   194→      const t = setTimeout(() => {
   195→        exitFullscreen();
   196→        setNavigationLocked(false);
   197→        proctorRef.current.stopMonitoring();
   198→        set("candidate-coding-result");
   199→      }, 1500);
   200→      return () => clearTimeout(t);
   201→    }
   202→     
   203→  }, [proctor.strikes]);
   204→
   205→  // Auto-submit -> navigate
   206→  useEffect(() => {
   207→    if (autoSubmitted) {
   208→      const t = setTimeout(() => {
   209→        exitFullscreen();
   210→        setNavigationLocked(false);
   211→        proctorRef.current.stopMonitoring();
   212→        set("candidate-coding-result");
   213→      }, 1200);
   214→      return () => clearTimeout(t);
   215→    }
   216→     
   217→  }, [autoSubmitted]);
   218→
   219→  const handleSubmitRound = () => {
   220→    exitFullscreen();
   221→    setNavigationLocked(false);
   222→    proctorRef.current.stopMonitoring();
   223→    setSubmitOpen(false);
   224→    set("candidate-coding-result");
   225→  };
   226→
   227→  const active = PROBLEMS[activeIdx];
   228→  const activeState = states[active.id];
   229→
   230→  const setCode = (code: string) => {
   231→    setStates((p) => ({
   232→      ...p,
   233→      [active.id]: { ...p[active.id], code: { ...p[active.id].code, [language]: code } },
   234→    }));
   235→  };
   236→

   239→    setTimeout(() => {
   240→      const results: TestCaseResult[] = active.tests.map((tc, i) => ({
   241→        id: tc.id,
   242→        status: (i === 2 ? "fail" : "pass") as "pass" | "fail",
   243→        runtime: `${8 + Math.floor(Math.random() * 12)}ms`,
   244→        got: i === 2 ? "wrong answer" : undefined,
   245→      }));
   246→      setStates((p) => ({
   247→        ...p,
   248→        [active.id]: { ...p[active.id], running: false, results },
   249→      }));
   250→    }, 900);
   251→  };
   252→
   253→  const submitProblem = () => {
   254→    setStates((p) => ({ ...p, [active.id]: { ...p[active.id], running: true } }));

   257→        id: tc.id,
   258→        status: (i === 2 ? "fail" : "pass") as "pass" | "fail",
   259→        runtime: `${8 + Math.floor(Math.random() * 12)}ms`,
   260→        got: i === 2 ? "wrong answer" : undefined,
   261→      }));
   262→      setStates((p) => ({
   263→        ...p,
   264→        [active.id]: { ...p[active.id], running: false, results, submitted: true },
   265→      }));
   266→    }, 1200);
   267→  };
   268→
   269→  // Status per problem
   270→  const problemStatus = (p: Problem): ProblemStatus => {

   273→    if (s.results || s.submitted) return "attempted";
   274→    return "unsolved";
   275→  };
   276→
   277→  const solvedCount = PROBLEMS.filter((p) => problemStatus(p) === "solved").length;
   278→  const attemptedCount = PROBLEMS.filter((p) => problemStatus(p) === "attempted").length;
   279→
   280→  const mins = Math.floor(timeLeft / 60);
   281→  const secs = timeLeft % 60;
   282→  const pad = (n: number) => String(n).padStart(2, "0");
   283→  const lowTime = timeLeft < 300;
   284→
   285→  const integrity =
   286→    proctor.strikes === 0
   287→      ? { label: "Clean", tone: "text-emerald-600 dark:text-emerald-300" }
   288→      : proctor.strikes < 3
   289→        ? { label: "Flags", tone: "text-amber-600 dark:text-amber-300" }
   290→        : { label: "Penalty", tone: "text-rose-600 dark:text-rose-300" };
   291→
   292→  const tabSwitches = proctor.events.filter((e) => e.type === "tab-switch" || e.type === "visibility-change").length;
   293→
   294→  return (
   295→    <div className="min-h-screen bg-background text-foreground">
   296→      <section className="mx-auto w-full max-w-[1400px] px-4 pb-32 pt-6 sm:px-6 lg:px-8">
   297→        {/* Header */}
   298→        <motion.div
   299→          initial={reduce ? false : { opacity: 0, y: 12 }}
   300→          animate={{ opacity: 1, y: 0 }}
   301→          transition={{ duration: 0.4 }}
   302→          className="flex flex-col gap-4 sm:flex-row sm:items-center sm:justify-between"

   304→          <div>
   305→            <Badge variant="outline" className="mb-2 rounded-full border-amber-400/40 bg-amber-500/10 px-3 py-1 text-amber-700 dark:text-amber-300">
   306→              <ShieldAlert className="mr-1 h-3 w-3" />

   312→            <p className="mt-1 text-xs text-muted-foreground">
   313→              {PROBLEMS.length} problems · 60 minutes · one submission per problem
   314→            </p>
   315→          </div>
   316→          <div className={cn(
   317→            "flex items-center gap-2 rounded-lg border px-4 py-2 font-mono text-lg font-semibold tabular-nums sm:text-2xl",
   318→            lowTime
   319→              ? "border-rose-400/40 bg-rose-500/10 text-rose-600 dark:text-rose-300"
   320→              : "border-amber-400/40 bg-amber-500/10 text-amber-600 dark:text-amber-300",
   321→          )}>
   322→            <Clock className="h-4 w-4" />
   323→            {pad(mins)}:{pad(secs)}
   324→          </div>
   325→        </motion.div>
   326→
   327→        <div className="mt-5 grid grid-cols-1 gap-5 lg:grid-cols-[260px_1fr]">
   328→          {/* Sidebar: problem list */}
   329→          <aside className="space-y-4">
   330→            <div className="rounded-lg border border-black/10 bg-card p-4 shadow-sm dark:border-white/10">
   331→              <div className="mb-3 flex items-center gap-2 text-xs uppercase tracking-wider text-muted-foreground">
   332→                <ListChecks className="h-3.5 w-3.5" />
   333→                Problems
   334→              </div>
   335→              <ul className="space-y-1.5">
   336→                {PROBLEMS.map((p, i) => {
   337→                  const status = problemStatus(p);
   338→                  const isActive = i === activeIdx;
   339→                  return (
   340→                    <li key={p.id}>
   341→                      <button
   342→                        onClick={() => setActiveIdx(i)}
   343→                        className={cn(
   344→                          "flex w-full items-center gap-2.5 rounded-md border px-3 py-2.5 text-left text-sm transition-colors",
   345→                          isActive
   346→                            ? "border-violet-500 bg-violet-500/10"
   347→                            : "border-black/10 bg-black/[0.02] hover:border-violet-400/40 dark:border-white/10 dark:bg-white/[0.02]",
   348→                        )}
   349→                      >
   350→                        <span className={cn(
   351→                          "grid h-6 w-6 shrink-0 place-items-center rounded-full text-[10px] font-semibold",
   352→                          status === "solved"
   353→                            ? "bg-emerald-500/15 text-emerald-600 dark:text-emerald-300"
   354→                            : status === "attempted"
   355→                              ? "bg-amber-500/15 text-amber-600 dark:text-amber-300"
   356→                              : "bg-black/5 text-muted-foreground dark:bg-white/10",
   357→                        )}>
   358→                          {status === "solved" ? (
   359→                            <CheckCircle2 className="h-3.5 w-3.5" />
   360→                          ) : status === "attempted" ? (
   361→                            <AlertTriangle className="h-3.5 w-3.5" />
   362→                          ) : (
   363→                            <span>{i + 1}</span>
   364→                          )}
   365→                        </span>
   366→                        <div className="min-w-0 flex-1">
   367→                          <p className={cn("truncate text-sm", isActive ? "font-semibold" : "font-medium")}>
   368→                            {p.title}
   369→                          </p>
   370→                          <p className="text-[10px] uppercase tracking-wider text-muted-foreground">
   371→                            {status === "solved" ? "Solved" : status === "attempted" ? "Attempted" : "Unsolved"} · {p.difficulty}
   372→                          </p>
   373→                        </div>
   374→                      </button>
   375→                    </li>
   376→                  );
   377→                })}
   378→              </ul>
   379→              <Separator className="my-3 bg-black/5 dark:bg-white/5" />
   380→              <div className="grid grid-cols-3 gap-2 text-center text-[10px]">
   381→                <div>
   382→                  <p className="font-mono text-sm font-semibold text-emerald-600 dark:text-emerald-300">{solvedCount}</p>
   383→                  <p className="uppercase tracking-wider text-muted-foreground">Solved</p>
   384→                </div>
   385→                <div>
   386→                  <p className="font-mono text-sm font-semibold text-amber-600 dark:text-amber-300">{attemptedCount}</p>
   387→                  <p className="uppercase tracking-wider text-muted-foreground">Attempted</p>
   388→                </div>
   389→                <div>
   390→                  <p className="font-mono text-sm font-semibold text-muted-foreground">{PROBLEMS.length - solvedCount - attemptedCount}</p>
   391→                  <p className="uppercase tracking-wider text-muted-foreground">Unsolved</p>
   392→                </div>
   393→              </div>
   394→            </div>
   395→
   396→            <Button
   397→              onClick={() => setSubmitOpen(true)}
   398→              className="w-full rounded-full bg-violet-500 text-white hover:bg-violet-600"
   399→            >
   400→              <CheckCircle2 className="mr-1.5 h-4 w-4" />
   401→              Submit Round
   402→            </Button>
   403→          </aside>
   404→
   405→          {/* Right: problem statement + editor + tests */}
   406→          <div className="space-y-5">
   407→            {/* Problem statement */}
   408→            <motion.div
   409→              key={active.id}
   410→              initial={reduce ? false : { opacity: 0, y: 8 }}
   411→              animate={{ opacity: 1, y: 0 }}
   412→              transition={{ duration: 0.25 }}
   413→              className="rounded-lg border border-black/10 bg-card p-5 shadow-sm sm:p-6 dark:border-white/10"
   414→            >
   415→              <div className="flex items-center justify-between">
   416→                <h2 className="text-lg font-semibold tracking-tight">
   417→                  <span className="font-mono text-muted-foreground">#{active.id}</span> {active.title}
   418→                </h2>
   419→                <Badge className={cn(
   420→                  "border",
   421→                  active.difficulty === "Easy"
   422→                    ? "border-emerald-400/30 bg-emerald-500/10 text-emerald-700 dark:text-emerald-300"
   423→                    : active.difficulty === "Medium"
   424→                      ? "border-amber-400/30 bg-amber-500/10 text-amber-700 dark:text-amber-300"
   425→                      : "border-rose-400/30 bg-rose-500/10 text-rose-700 dark:text-rose-300",
   426→                )}>
   427→                  {active.difficulty}
   428→                </Badge>
   429→              </div>
   430→              <p className="mt-3 text-sm text-foreground/85">
   431→                {active.description}
   432→              </p>
   433→              <div className="mt-4">
   434→                <p className="text-xs uppercase tracking-wider text-muted-foreground">Examples</p>
   435→                <div className="mt-2 space-y-2">
   436→                  {active.tests.filter((t) => !t.hidden).map((tc) => (
   437→                    <div key={tc.id} className="rounded-md border border-black/10 bg-zinc-900 p-3 font-mono text-xs dark:border-white/10">
   438→                      <p className="text-emerald-300">Input: <span className="text-white/80">{tc.input}</span></p>
   439→                      <p className="mt-1 text-violet-300">Output: <span className="text-white/80">{tc.expected}</span></p>
   440→                    </div>
   441→                  ))}
   442→                </div>
   443→              </div>
   444→            </motion.div>
   445→
   446→            {/* Editor */}
   447→            <div className="overflow-hidden rounded-lg border border-black/10 bg-card shadow-sm dark:border-white/10">
   448→              {/* Editor chrome */}
   449→              <div className="flex items-center justify-between border-b border-black/10 px-4 py-2 dark:border-white/5">
   450→                <span className="flex items-center gap-1.5 font-mono text-xs text-muted-foreground">
   451→                  <Terminal className="h-3 w-3" />
   452→                  solution.{language === "Python" ? "py" : "js"}
   453→                </span>
   454→                <div className="flex items-center gap-2">
   455→                  <Select value={language} onValueChange={setLanguage}>
   456→                    <SelectTrigger size="sm" className="w-32 rounded-md border-black/10 bg-black/[0.02] dark:border-white/10 dark:bg-white/5">
   457→                      <SelectValue />
   458→                    </SelectTrigger>
   459→                    <SelectContent>
   460→                      {LANGUAGES.map((l) => (
   461→                        <SelectItem key={l} value={l}>{l}</SelectItem>
   462→                      ))}
   463→                    </SelectContent>
   464→                  </Select>
   465→                </div>
   466→              </div>
   467→
   468→              {/* Code area */}
   469→              <div className="relative bg-zinc-900">
   470→                <div className="flex">
   471→                  <div className="select-none border-r border-white/5 px-2 py-3 text-right font-mono text-xs text-white/30">
   472→                    {(activeState.code[language] ?? "").split("\n").map((_, i) => (
   473→                      <div key={i} className="leading-6">{i + 1}</div>
   474→                    ))}
   475→                  </div>
   476→                  <textarea
   477→                    value={activeState.code[language] ?? ""}
   478→                    onChange={(e) => setCode(e.target.value)}
   479→                    spellCheck={false}
   480→                    disabled={activeState.submitted}
   481→                    className="flex-1 resize-none bg-transparent p-3 font-mono text-xs leading-6 text-emerald-100 outline-none disabled:opacity-60"
   482→                    style={{ minHeight: 240, tabSize: 4 }}
   483→                  />
   484→                </div>
   485→              </div>
   486→
   487→              {/* Action bar */}
   488→              <div className="flex flex-wrap items-center gap-2 border-t border-black/10 p-4 dark:border-white/5">
   489→                <Button
   490→                  variant="outline"
   491→                  size="sm"
   492→                  onClick={runTests}
   493→                  disabled={activeState.running}
   494→                  className="rounded-full border-black/10 bg-black/[0.02] hover:bg-black/[0.05] dark:border-white/15 dark:bg-white/5 dark:hover:bg-white/10"

   503→                  className="rounded-full bg-violet-500 text-white hover:bg-violet-600 disabled:opacity-50"

   507→                </Button>
   508→                {activeState.submitted && (

   510→                    Submitted. No retry allowed.
   511→                  </span>

   513→              </div>
   514→
   515→              {/* Test results */}
   516→              {activeState.results && (
   517→                <div className="border-t border-black/10 p-4 dark:border-white/5">
   518→                  <div className="mb-3 flex items-center justify-between">
   519→                    <p className="text-xs uppercase tracking-wider text-muted-foreground">Test Results</p>
   520→                    <span className={cn(
   521→                      "text-xs font-semibold",
   522→                      activeState.results.every((r) => r.status === "pass")
   523→                        ? "text-emerald-600 dark:text-emerald-300"
   524→                        : "text-rose-600 dark:text-rose-300",
   525→                    )}>
   526→                      {activeState.results.filter((r) => r.status === "pass").length}/{activeState.results.length} passed
   527→                    </span>
   528→                  </div>
   529→                  <ul className="space-y-1.5">
   530→                    {activeState.results.map((r) => {
   531→                      const tc = active.tests.find((t) => t.id === r.id);
   532→                      return (
   533→                        <li
   534→                          key={r.id}
   535→                          className={cn(
   536→                            "flex items-center gap-3 rounded-md border px-3 py-2 text-xs",
   537→                            r.status === "pass"
   538→                              ? "border-emerald-400/30 bg-emerald-500/5"
   539→                              : "border-rose-400/30 bg-rose-500/5",
   540→                          )}
   541→                        >
   542→                          {r.status === "pass" ? (
   543→                            <CheckCircle2 className="h-3.5 w-3.5 text-emerald-500" />
   544→                          ) : (
   545→                            <XCircle className="h-3.5 w-3.5 text-rose-500" />
   546→                          )}
   547→                          <span className="flex-1 font-mono text-foreground/80">
   548→                            Case {r.id}{tc?.hidden ? " (hidden)" : ""}: {tc?.input}
   549→                          </span>
   550→                          <span className="font-mono text-muted-foreground">{r.runtime}</span>
   551→                          {r.got && <span className="font-mono text-rose-600 dark:text-rose-300">got: {r.got}</span>}
   552→                        </li>
   553→                      );
   554→                    })}
   555→                  </ul>
   556→                </div>
   557→              )}
   558→            </div>
   559→
   560→            {/* Problem navigation */}
   561→            <div className="flex items-center justify-between">
   562→              <Button
   563→                variant="outline"
   564→                size="sm"
   565→                onClick={() => setActiveIdx((i) => Math.max(0, i - 1))}
   566→                disabled={activeIdx === 0}
   567→                className="rounded-full border-black/10 bg-black/[0.02] hover:bg-black/[0.05] disabled:opacity-40 dark:border-white/15 dark:bg-white/5 dark:hover:bg-white/10"
   568→              >
   569→                <ChevronLeft className="mr-1 h-4 w-4" />
   570→                Previous Problem
   571→              </Button>
   572→              <span className="text-xs text-muted-foreground">
   573→                Problem {activeIdx + 1} of {PROBLEMS.length}
   574→              </span>
   575→              {activeIdx === PROBLEMS.length - 1 ? (
   576→                <Button
   577→                  size="sm"
   578→                  onClick={() => setSubmitOpen(true)}
   579→                  className="rounded-full bg-violet-500 text-white hover:bg-violet-600"
   580→                >
   581→                  Submit Round
   582→                  <ChevronRight className="ml-1 h-4 w-4" />
   583→                </Button>
   584→              ) : (
   585→                <Button
   586→                  size="sm"
   587→                  onClick={() => setActiveIdx((i) => Math.min(PROBLEMS.length - 1, i + 1))}
   588→                  className="rounded-full bg-violet-500 text-white hover:bg-violet-600"
   589→                >
   590→                  Next Problem
   591→                  <ChevronRight className="ml-1 h-4 w-4" />
   592→                </Button>
   593→              )}
   594→            </div>
   595→          </div>
   596→        </div>
   597→
   598→        {/* Proctoring footer */}
   599→        <div className="mt-6 flex flex-col items-start justify-between gap-2 rounded-lg border border-black/10 bg-card px-4 py-3 text-xs shadow-sm sm:flex-row sm:items-center dark:border-white/10">
   600→          <div className="flex flex-wrap items-center gap-x-4 gap-y-1 text-muted-foreground">
   601→            <span>Tab switches: <span className="font-semibold text-foreground">{tabSwitches}</span></span>
   602→            <span>Strikes: <span className="font-semibold text-foreground">{proctor.strikes}/3</span></span>
   603→            <span className="flex items-center gap-1">
   604→              Integrity:
   605→              <span className={cn("flex items-center gap-1 font-semibold", integrity.tone)}>
   606→                {proctor.strikes === 0 ? <ShieldCheck className="h-3 w-3" /> : <AlertTriangle className="h-3 w-3" />}
   607→                {integrity.label}
   608→              </span>
   609→            </span>
   610→          </div>
   611→          <p className="text-[11px] text-muted-foreground">
   612→            Each problem allows one submission. Three strikes will auto-submit your round.
   613→          </p>
   614→        </div>
   615→      </section>
   616→
   617→      {/* Strike warning */}
   618→      <ProctoringWarning
   619→        open={proctor.strikes > 0 && proctor.strikes < 3}
   620→        strikeNumber={proctor.strikes}
   621→        violationType={proctor.events[proctor.events.length - 1]?.type ?? "tab-switch"}
   622→        onAcknowledge={() => proctor.reset()}
   623→        onAutoSubmit={() => setAutoSubmitted(true)}
   624→      />
   625→
   626→      {/* Submit round confirmation */}
   627→      <Dialog open={submitOpen} onOpenChange={setSubmitOpen}>
   628→        <DialogContent className="border-black/10 bg-card dark:border-white/10">
   629→          <DialogHeader>
   630→            <DialogTitle>Submit the coding round?</DialogTitle>
   631→            <DialogDescription>
   632→              You have solved {solvedCount} of {PROBLEMS.length} problems and attempted {attemptedCount}.
   633→              This action cannot be undone. All unsaved progress on unsubmitted problems will be lost.
   634→            </DialogDescription>
   635→          </DialogHeader>
   636→          <DialogFooter className="mt-4 flex gap-2">
   637→            <Button variant="ghost" onClick={() => setSubmitOpen(false)} className="rounded-full">
   638→              Keep Going
   639→            </Button>
   640→            <Button
   641→              onClick={handleSubmitRound}
   642→              className="rounded-full bg-violet-500 text-white hover:bg-violet-600"
   643→            >
   644→              <CheckCircle2 className="mr-1.5 h-4 w-4" />
   645→              Submit Round
   646→              <ArrowRight className="ml-1.5 h-4 w-4" />
   647→            </Button>
   648→          </DialogFooter>
   649→        </DialogContent>
   650→      </Dialog>
   651→
   652→      {/* Auto-submit overlay */}
   653→      <AnimatePresence>
   654→        {autoSubmitted && (
   655→          <motion.div
   656→            initial={{ opacity: 0 }}
   657→            animate={{ opacity: 1 }}
   658→            exit={{ opacity: 0 }}
   659→            className="fixed inset-0 z-[80] grid place-items-center bg-background/80 backdrop-blur-sm"
   660→          >
   661→            <motion.div
   662→              initial={reduce ? false : { scale: 0.9, opacity: 0 }}
   663→              animate={{ scale: 1, opacity: 1 }}
   664→              className="mx-4 max-w-sm rounded-lg border border-black/10 bg-card p-6 text-center shadow-lg dark:border-white/10"
   665→            >
   666→              {proctor.strikes >= 3 ? (
   667→                <AlertTriangle className="mx-auto h-10 w-10 text-rose-500" />
   668→              ) : (
   669→                <XCircle className="mx-auto h-10 w-10 text-amber-500" />
   670→              )}
   671→              <h3 className="mt-3 text-lg font-semibold tracking-tight">
   672→                {proctor.strikes >= 3 ? "Auto-submitted" : "Time up"}
   673→              </h3>
   674→              <p className="mt-1 text-xs text-muted-foreground">
   675→                {proctor.strikes >= 3
   676→                  ? "Three proctoring strikes. Your round has been submitted automatically."
   677→                  : "The 60-minute timer has elapsed. Submitting your round now."}
   678→              </p>
   679→            </motion.div>
   680→          </motion.div>
   681→        )}
   682→      </AnimatePresence>
   683→    </div>
   684→  );
   685→}
