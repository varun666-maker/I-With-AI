     1→/**
     2→ * Mock question bank for the Coding Arena. Pure data module, no React,
     3→ * so it can be imported from server components, API routes, or client
     4→ * components without triggering any bundler restrictions.
     5→ *
     6→ * 24 problems are spread across 6 topics:
     7→ *   - Arrays (5)   - Strings (4)   - DP (5)
     8→ *   - Graphs (3)    - SQL (3)       - Misc (4)
     9→ *
    10→ * Every problem ships with 10 test cases (3 visible + 7 hidden) plus
    11→ * Python and JavaScript starter code.
    12→ */
    13→
    14→export type Difficulty = "Easy" | "Medium" | "Hard";
    15→
    16→export type Topic = "Arrays" | "Strings" | "DP" | "Graphs" | "SQL" | "Misc";
    17→
    18→export const TOPICS: Topic[] = ["Arrays", "Strings", "DP", "Graphs", "SQL", "Misc"];
    19→export const DIFFICULTIES: Difficulty[] = ["Easy", "Medium", "Hard"];
    20→
    21→export interface CodingExample {
    22→  input: string;
    23→  output: string;
    24→  explanation?: string;
    25→}
    26→
    27→export interface TestCase {
    28→  id: number;
    29→  input: string;
    30→  expected: string;
    31→  hidden: boolean;
    32→}
    33→
    34→export interface CodingProblem {
    35→  id: number;
    36→  title: string;
    37→  difficulty: Difficulty;
    38→  topic: Topic;
    39→  description: string;
    40→  examples: CodingExample[];
    41→  constraints: string[];
    42→  testCases: TestCase[];
    43→  functionSignature: string;
    44→  starterCode: Record<string, string>;
    45→  acceptance: number;
    46→}
    47→
    48→const PY_STARTER = "def solution(input):\n    pass\n";
    49→const JS_STARTER = "function solution(input) {\n  // Write your solution here\n}\n";
    50→
    51→function py(fnName: string): string {
    52→  return `def ${fnName}(input):\n    pass\n`;
    53→}
    54→
    55→function js(fnName: string): string {
    56→  return `function ${fnName}(input) {\n  // Write your solution here\n}\n`;
    57→}
    58→
    59→function starter(fnName: string): Record<string, string> {
    60→  return { Python: py(fnName), JavaScript: js(fnName) };
    61→}
    62→
    63→/**
    64→ * Convenience factory: builds the 10-case test array with the first 3
    65→ * marked visible and the remaining 7 marked hidden.
    66→ */
    67→function cases(rows: Array<[string, string]>): TestCase[] {
    68→  return rows.map(([input, expected], i) => ({
    69→    id: i + 1,
    70→    input,
    71→    expected,
    72→    hidden: i >= 3,
    73→  }));
    74→}
