   551→                      setSeconds(30 * 60);

   553→                    className="rounded-full border-black/10 bg-black/[0.02] hover:bg-black/[0.05] dark:border-white/15 dark:bg-white/5 dark:hover:bg-white/10"

   563→                </div>

   566→          </AnimatePresence>
   567→        </div>
