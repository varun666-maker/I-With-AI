   518→  const removeEdu = (i: number) => setEducation(education.filter((_, idx) => idx !== i));
   519→  const updateEdu = (i: number, key: "degree" | "institution" | "year", val: string) =>

   711→function LoginPanel({ onLogin }: { onLogin: () => void }) {
   712→  const [email, setEmail] = useState("");
   713→  const [pwd, setPwd] = useState("");
   714→  const [loading, setLoading] = useState(false);
   715→  const [forgot, setForgot] = useState(false);
   716→  const [resetSent, setResetSent] = useState(false);
   717→  const [showPwd, setShowPwd] = useState(false);
   718→  const [errors, setErrors] = useState<{ email?: string; pwd?: string }>({});
   719→  const [touched, setTouched] = useState<{ email?: boolean; pwd?: boolean }>({});
   720→
   721→  const validateEmail = (v: string) => {
   722→    if (!v) return "Email is required";
   723→    if (!/^[^\s@]+@[^\s@]+\.[^\s@]+$/.test(v)) return "Enter a valid email address";
   724→    return undefined;
   725→  };
   726→  const validatePwd = (v: string) => {
   727→    if (!v) return "Password is required";
   728→    if (v.length < 8) return "Password must be at least 8 characters";
   729→    return undefined;
   730→  };
   731→
   732→  const valid = !validateEmail(email) && !validatePwd(pwd);
   733→
   734→  const login = () => {
   735→    const newErrors = { email: validateEmail(email), pwd: validatePwd(pwd) };
   736→    setErrors(newErrors);
   737→    setTouched({ email: true, pwd: true });
   738→    if (newErrors.email || newErrors.pwd) return;
   739→    setLoading(true);
   740→    setTimeout(() => {
   741→      setLoading(false);
   742→      onLogin();
   743→    }, 900);
   744→  };
   745→
   746→  return (
   747→    <div>
   748→      <h3 className="text-lg font-semibold">Welcome back</h3>
   749→      <p className="mt-1 text-sm text-muted-foreground">Log in to continue your prep journey.</p>
   750→
   751→      {!forgot ? (
   752→        <>
   753→          <div className="mt-5 space-y-3">
   754→            <div>
   755→              <Label className="text-xs text-muted-foreground">Email Address</Label>
   756→              <div className="relative mt-1">
   757→                <Mail className="absolute left-3 top-1/2 h-4 w-4 -translate-y-1/2 text-muted-foreground" />
   758→                <Input
   759→                  type="email"
   760→                  value={email}
   761→                  onChange={(e) => {
   762→                    setEmail(e.target.value);
   763→                    if (touched.email) setErrors((p) => ({ ...p, email: validateEmail(e.target.value) }));
   764→                  }}
   765→                  onBlur={() => {
   766→                    setTouched((p) => ({ ...p, email: true }));
   767→                    setErrors((p) => ({ ...p, email: validateEmail(email) }));
   768→                  }}
   769→                  placeholder="you@email.com"
   770→                  aria-invalid={!!errors.email && touched.email}
   771→                  aria-describedby={errors.email && touched.email ? "login-email-error" : undefined}
   772→                  className={cn(
   773→                    "rounded-lg bg-black/[0.02] pl-9 border-black/10 dark:bg-white/5 dark:border-white/10",
   774→                    errors.email && touched.email && "border-rose-400/60 focus-visible:ring-rose-400/30"
   775→                  )}
   776→                />
   777→              </div>
   778→              {errors.email && touched.email && (
   779→                <p id="login-email-error" className="mt-1 text-xs text-rose-500 dark:text-rose-400" role="alert">
   780→                  {errors.email}
   781→                </p>
   782→              )}
   783→            </div>
   784→            <div>
   785→              <Label className="text-xs text-muted-foreground">Password</Label>
   786→              <div className="relative mt-1">
   787→                <Lock className="absolute left-3 top-1/2 h-4 w-4 -translate-y-1/2 text-muted-foreground" />
   788→                <Input
   789→                  type={showPwd ? "text" : "password"}
   790→                  value={pwd}
   791→                  onChange={(e) => {
   792→                    setPwd(e.target.value);
   793→                    if (touched.pwd) setErrors((p) => ({ ...p, pwd: validatePwd(e.target.value) }));
   794→                  }}
   795→                  onBlur={() => {
   796→                    setTouched((p) => ({ ...p, pwd: true }));
   797→                    setErrors((p) => ({ ...p, pwd: validatePwd(pwd) }));
   798→                  }}
   799→                  placeholder="••••••••"
   800→                  aria-invalid={!!errors.pwd && touched.pwd}
   801→                  aria-describedby={errors.pwd && touched.pwd ? "login-pwd-error" : undefined}
   802→                  className={cn(
   803→                    "rounded-lg bg-black/[0.02] pl-9 pr-9 border-black/10 dark:bg-white/5 dark:border-white/10",
   804→                    errors.pwd && touched.pwd && "border-rose-400/60 focus-visible:ring-rose-400/30"
   805→                  )}
   806→                />
   807→                <button
   808→                  type="button"
   809→                  onClick={() => setShowPwd((s) => !s)}
   810→                  className="absolute right-2 top-1/2 -translate-y-1/2 rounded-md p-1 text-muted-foreground hover:text-foreground"
   811→                  aria-label={showPwd ? "Hide password" : "Show password"}

   814→                </button>

   816→              {errors.pwd && touched.pwd && (

   818→                  {errors.pwd}

   843→            {loading ? (

   845→                <Loader2 className="mr-2 h-4 w-4 animate-spin" /> Signing in…

   849→                Login <ArrowRight className="ml-2 h-4 w-4" />

   852→          </Button>

   856→          {!resetSent ? (
   857→            <motion.div
   858→              key="forgot-form"
   859→              initial={{ opacity: 0, y: 8 }}
