   461→          dragging

   464→        )}
   465→      >

   471→          >
   472→            <Upload className="h-6 w-6" />
   473→          </motion.div>
   474→          <p className="mt-4 text-sm font-medium">
   475→            {dragging ? "Drop your file to upload" : "Drag & drop your resume"}
   476→          </p>

   478→            or <span className="text-violet-600 underline-offset-2 group-hover:underline dark:text-violet-400">browse files</span> · PDF or DOCX · max 5&nbsp;MB
   479→          </p>
   480→        </div>
   481→      </button>
   482→
   483→      {file && (
   484→        <div className="mt-3 flex items-center justify-between rounded-lg border border-black/10 bg-black/[0.02] px-3 py-2.5 dark:border-white/10 dark:bg-white/[0.03]">
   485→          <div className="flex min-w-0 items-center gap-3">

   488→            </span>

   490→              <p className="truncate text-sm font-medium">{file.name}</p>

   492→            </div>
   493→          </div>

   495→            <button
   496→              onClick={onClear}
   497→              className="grid h-7 w-7 place-items-center rounded-md text-muted-foreground hover:text-rose-500"
   498→              aria-label="Remove file"
   499→            >
   500→              <X className="h-4 w-4" />
   501→            </button>
   502→            <Button
   503→              onClick={onAnalyze}
   504→              disabled={loading}
   505→              className="rounded-full bg-violet-500 text-white hover:bg-violet-600"
   506→            >
   507→              {loading ? (
   508→                <>
   509→                  <Loader2 className="mr-2 h-4 w-4 animate-spin" /> Analyzing…
   510→                </>
   511→              ) : (

   514→                </>
   515→              )}
   516→            </Button>
   517→          </div>
   518→        </div>
   519→      )}
   520→    </div>
   521→  );
   522→}

   524→function ScoreCard({
   525→  label,
   526→  value,
   527→  max,
   528→  tone,
   529→  hint,
   530→}: {
   531→  label: string;
   532→  value: number;
   533→  max: number;
   534→  tone: "emerald" | "violet" | "amber";
   535→  hint: string;
   536→}) {
   537→  const toneRing: Record<string, string> = {
   538→    emerald: "text-emerald-600 dark:text-emerald-300",
   539→    violet: "text-violet-500 dark:text-violet-300",
   540→    amber: "text-amber-600 dark:text-amber-300",
   541→  };
   542→  const pct = (value / max) * 100;
   543→  return (
   544→    <div className="rounded-lg border border-black/10 bg-card p-6 shadow-sm dark:border-white/10">
   545→      <p className="text-xs uppercase tracking-wider text-muted-foreground">{label}</p>
   546→      <div className="mt-2 flex items-baseline gap-1">
   547→        <span className={cn("text-4xl font-semibold tracking-tight", toneRing[tone])}>{value}</span>
   548→        <span className="text-sm text-muted-foreground">/{max}</span>
   549→      </div>
   550→      <Progress value={pct} className={cn("mt-3 bg-black/10 dark:bg-white/10")} />
   551→      <p className="mt-2 text-[11px] text-muted-foreground">{hint}</p>
   552→    </div>
   553→  );
   554→}
