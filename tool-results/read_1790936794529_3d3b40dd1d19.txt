    67→/**
    68→ * CodingArena
    69→ *
    70→ * Prep-mode coding practice. Splits into two views driven by the
    71→ * `codingView` flag in the global nav store:
    72→ *
    73→ *   - "list"   -> LeetCode-style problem index (ProblemListTable) with a
    74→ *                 submission heatmap, topic progress, and a random-picker.
    75→ *   - "editor" -> VS Code-like split view (problem statement + Monaco-styled
    76→ *                 textarea editor) with fullscreen + navigation lock.
    77→ *
    78→ * Prep mode is forgiving: exiting fullscreen via Esc shows a friendly
    79→ * "Re-enter fullscreen" modal with no penalty. Submissions are recorded
    80→ * to the store so the heatmap and solved/attempted counters update live.
    81→ */

    87→  Medium: "border-amber-400/30 bg-amber-500/10 text-amber-700 dark:text-amber-200",
    88→  Hard: "border-rose-400/30 bg-rose-500/10 text-rose-700 dark:text-rose-200",

    91→interface TestCaseResult {
    92→  id: number;
    93→  status: "pass" | "fail";
    94→  runtime: string;
    95→  mem: string;
    96→  got?: string;

    99→/**
   100→ * Pick a random problem the candidate hasn't solved yet. If everything is
   101→ * solved, fall back to any problem that isn't the one they're currently on.
   102→ */
   103→function pickRandomUnsolved(solvedIds: Set<number>, excludeId?: number): CodingProblem {
   104→  const candidates = CODING_PROBLEMS.filter(
   105→    (p) => !solvedIds.has(p.id) && p.id !== excludeId,
   106→  );
   107→  if (candidates.length > 0) {
   108→    return candidates[Math.floor(Math.random() * candidates.length)];
   109→  }
   110→  const fallback = CODING_PROBLEMS.filter((p) => p.id !== excludeId);
   111→  return fallback[Math.floor(Math.random() * fallback.length)] ?? CODING_PROBLEMS[0];
   112→}

   114→export function CodingArena() {
   115→  const codingView = useNav((s) => s.codingView);
   116→  if (codingView === "list") return <ProblemListView />;
   117→  return <CodingEditorView />;
   118→}

   124→function ProblemListView() {
   125→  const solvedProblems = useNav((s) => s.solvedProblems);
   126→  const attemptedProblems = useNav((s) => s.attemptedProblems);
   127→  const submissions = useNav((s) => s.submissions);
   128→  const setCodingView = useNav((s) => s.setCodingView);
   129→  const setCurrentProblem = useNav((s) => s.setCurrentProblem);
   130→  const setNavigationLocked = useNav((s) => s.setNavigationLocked);
   131→
   132→  // The store persists solved/attempted IDs as strings (localStorage friendly),
   133→  // but ProblemListTable works with numeric IDs. Convert once per render.
   134→  const solvedNums = useMemo(() => solvedProblems.map((id) => Number(id)), [solvedProblems]);
   135→  const attemptedNums = useMemo(
   136→    () => attemptedProblems.map((id) => Number(id)),
   137→    [attemptedProblems],
   138→  );
   139→  const heatSubs = useMemo(
   140→    () => submissions.map((s) => ({ ...s, problemId: Number(s.problemId) })),
   141→    [submissions],
   142→  );
   143→
   144→  const enterProblem = (p: CodingProblem) => {
   145→    setCurrentProblem(String(p.id));
   146→    setCodingView("editor");
   147→    requestFullscreen();
   148→    setNavigationLocked(true);
   149→  };
   150→
   151→  const handleRandom = () => {
   152→    const solved = new Set(solvedNums);
   153→    enterProblem(pickRandomUnsolved(solved));
   154→  };
   155→
   156→  return (
   157→    <div className="min-h-screen bg-background text-foreground">
   158→      <PageWrap wide>
   159→        <BackToHome />
   160→        <SectionHeading
   161→          eyebrow="Coding Arena"
   162→          title={
   163→            <>
   164→              Practice with the{" "}
   165→              <span className="text-violet-500 dark:text-violet-400">AI code reviewer</span>
   166→            </>
   167→          }
   168→          description="A VS Code-like editor with syntax highlighting, 10 test cases per problem, real-time execution, and an AI review that scores complexity, readability, and gives refactor suggestions."
   169→          align="left"
   170→        />
   171→
   172→        <div className="mt-6">
   173→          <ProblemListTable
   174→            solvedProblems={solvedNums}
   175→            attemptedProblems={attemptedNums}
   176→            submissions={heatSubs}
   177→            onPickProblem={enterProblem}
   178→            onRandomQuestion={handleRandom}
   179→          />
   180→        </div>
   181→      </PageWrap>
   182→    </div>
   183→  );
   184→}

   190→function CodingEditorView() {
   191→  const setCodingView = useNav((s) => s.setCodingView);
   192→  const currentProblemId = useNav((s) => s.currentProblemId);
   193→  const solvedProblems = useNav((s) => s.solvedProblems);
   194→  const setCurrentProblem = useNav((s) => s.setCurrentProblem);
   195→  const markSolved = useNav((s) => s.markSolved);
   196→  const markAttempted = useNav((s) => s.markAttempted);
   197→  const recordSubmission = useNav((s) => s.recordSubmission);
   198→  const setNavigationLocked = useNav((s) => s.setNavigationLocked);
   199→  const reduce = useReducedMotion();
   200→
   201→  // Resolve the active problem. Default to the first problem if the store
   202→  // doesn't have one yet (e.g. user deep-linked into the editor view).
   203→  const problem = useMemo<CodingProblem>(() => {
   204→    const id = currentProblemId ? Number(currentProblemId) : null;
   205→    if (id !== null) {
   206→      const found = getProblemById(id);
   207→      if (found) return found;
   208→    }
   209→    return CODING_PROBLEMS[0];
   210→  }, [currentProblemId]);
   211→
   212→  const [language, setLanguage] = useState<string>("Python");
   213→  const [code, setCode] = useState<string>(problem.starterCode.Python ?? "");
   214→  const [customInput, setCustomInput] = useState<string>(
   215→    problem.testCases[0]?.input ?? "",
   216→  );
   217→  const [results, setResults] = useState<TestCaseResult[] | null>(null);
   218→  const [submitted, setSubmitted] = useState(false);
   219→  const [running, setRunning] = useState(false);
   220→  const [confirmLang, setConfirmLang] = useState<string | null>(null);
   221→  const [autoSaved, setAutoSaved] = useState(false);
   222→  const [successOpen, setSuccessOpen] = useState(false);
   223→  const [reenterOpen, setReenterOpen] = useState(false);
   224→  const [fullscreenExits, setFullscreenExits] = useState(0);
   225→  const [tabSwitches] = useState(1);
   226→
   227→  // Reset the editor state whenever the active problem changes (e.g. when the
   228→  // candidate picks "Next Question"). Deferred to a microtask to keep the
   229→  // effect body free of synchronous setState calls.
   230→  useEffect(() => {
   231→    queueMicrotask(() => {
   232→      setLanguage("Python");
   233→      setCode(problem.starterCode.Python ?? "");
   234→      setCustomInput(problem.testCases[0]?.input ?? "");
   235→      setResults(null);
   236→      setSubmitted(false);
   237→    });
   238→  }, [problem]);
   239→
   240→  // Fullscreen lifecycle: request on mount, exit on unmount, and listen for
   241→  // the user pressing Esc so we can show a friendly "re-enter" modal. Prep
   242→  // mode is forgiving, so there are no penalties.
   243→  useEffect(() => {
   244→    requestFullscreen();
   245→    queueMicrotask(() => setNavigationLocked(true));
   246→    const unsubscribe = onFullscreenChange(() => {
   247→      if (!isCurrentlyFullscreen()) {
   248→        queueMicrotask(() => {
   249→          setReenterOpen(true);
   250→          setFullscreenExits((n) => n + 1);
   251→        });
   252→      }
   253→    });
   254→    return () => {
   255→      unsubscribe();
   256→      exitFullscreen();
   257→      queueMicrotask(() => setNavigationLocked(false));
   258→    };
   259→  }, [setNavigationLocked]);
   260→
   261→  // Autosave simulation. The indicator pulses every 30s to reassure the
   262→  // candidate their work is safe.
   263→  useEffect(() => {
   264→    const t = setInterval(() => {
   265→      queueMicrotask(() => setAutoSaved(true));
   266→      setTimeout(() => queueMicrotask(() => setAutoSaved(false)), 1500);
   267→    }, 30000);
   268→    return () => clearInterval(t);
   269→  }, []);
   270→
   271→  const onLangChange = (lang: string) => {
   272→    if (lang === language) return;
   273→    setConfirmLang(lang);
   274→  };
   275→
   276→  const confirmSwitch = () => {
   277→    if (!confirmLang) return;
   278→    setLanguage(confirmLang);
   279→    setCode(problem.starterCode[confirmLang] ?? "");
   280→    setResults(null);
   281→    setSubmitted(false);
   282→    setConfirmLang(null);
   283→  };
   284→
   285→  const run = () => {
   286→    setRunning(true);
   287→    setTimeout(() => {
   288→      setRunning(false);
   289→      // Mock: 8/10 pass. Cases at index 2 and 7 fail so the candidate sees
   290→      // realistic feedback before submitting.
   291→      setResults(
   292→        problem.testCases.map((tc, i) => ({
   293→          id: tc.id,
   294→          status: i === 2 || i === 7 ? ("fail" as const) : ("pass" as const),
   295→          runtime: `${8 + Math.floor(Math.random() * 12)}ms`,
   296→          mem: "14.0MB",
   297→          got: i === 2 ? "wrong answer" : i === 7 ? "timeout" : undefined,
   298→        })),
   299→      );
   300→    }, 900);
   301→  };
   302→
   303→  const submit = () => {
   304→    setRunning(true);
   305→    setTimeout(() => {
   306→      setRunning(false);
   307→      // Mock: all tests pass on submit (prep mode is forgiving). The failure
   308→      // path below stays in place so the success/attempted branching is real.
   309→      const newResults: TestCaseResult[] = problem.testCases.map((tc) => ({
   310→        id: tc.id,
   311→        status: "pass" as const,
   312→        runtime: `${8 + Math.floor(Math.random() * 12)}ms`,
   313→        mem: "14.0MB",
   314→      }));
   315→      setResults(newResults);
   316→      setSubmitted(true);
   317→      const allPassed = newResults.every((r) => r.status === "pass");
   318→      const idStr = String(problem.id);
   319→      if (allPassed) {
   320→        markSolved(idStr);
   321→        recordSubmission(idStr, true);
   322→        queueMicrotask(() => setSuccessOpen(true));
   323→      } else {
   324→        markAttempted(idStr);
   325→        recordSubmission(idStr, false);
   326→      }
   327→    }, 1400);
   328→  };
   329→
   330→  const passCount = results ? results.filter((r) => r.status === "pass").length : 0;
   331→
   332→  const handleBack = () => {
   333→    setCodingView("list");
   334→    exitFullscreen();
   335→    setNavigationLocked(false);
   336→  };
   337→
   338→  const handleNext = () => {
   339→    const solved = new Set(solvedProblems.map((id) => Number(id)));
   340→    const next = pickRandomUnsolved(solved, problem.id);
   341→    setCurrentProblem(String(next.id));
   342→    setSuccessOpen(false);
   343→  };
   344→
   345→  const reenterFullscreen = () => {
   346→    requestFullscreen();
   347→    setReenterOpen(false);
   348→  };
   349→
   350→  const difficultyBadge = DIFFICULTY_BADGE[problem.difficulty];
   351→  const fileExt = language === "Python" ? "py" : "js";
   352→
   353→  return (
   354→    <div className="min-h-screen bg-background text-foreground">
   355→      <PageWrap wide>
   356→        {/* Top bar: Back / Title / Next */}
   357→        <div className="flex flex-col gap-3 rounded-lg border border-black/10 bg-card p-3 shadow-sm sm:flex-row sm:items-center sm:justify-between sm:p-4 dark:border-white/10">
   358→          <Button
   359→            variant="outline"
   360→            size="sm"
   361→            onClick={handleBack}
   362→            className="rounded-full border-black/10 bg-black/[0.02] hover:bg-black/[0.05] dark:border-white/15 dark:bg-white/5 dark:hover:bg-white/10"
   363→          >
   364→            <ArrowLeft className="mr-2 h-4 w-4" /> Back to Problems
   365→          </Button>
   366→          <div className="flex min-w-0 flex-1 items-center justify-center gap-3 px-2 text-center">
   367→            <span className="truncate text-sm font-semibold tracking-tight sm:text-base">
   368→              <span className="font-mono text-muted-foreground">#{problem.id}</span>{" "}
   369→              {problem.title}
   370→            </span>
   371→            <Badge className={cn("shrink-0", difficultyBadge)}>{problem.difficulty}</Badge>
   372→          </div>
   373→          <Button
   374→            size="sm"
   375→            onClick={handleNext}
   376→            className="rounded-full bg-violet-500 text-white hover:bg-violet-600"
   377→          >
   378→            Next Question <ArrowRight className="ml-2 h-4 w-4" />
   379→          </Button>
   380→        </div>
   381→
   382→        <div className="mt-4 grid grid-cols-1 gap-4 lg:grid-cols-2">
   383→          {/* LEFT PANE: problem statement */}
   384→          <div className="flex h-full flex-col rounded-lg border border-black/10 bg-card p-6 shadow-sm dark:border-white/10">
   385→            <div className="flex items-center justify-between">
   386→              <div>
   387→                <h3 className="text-lg font-semibold tracking-tight">
   388→                  <span className="font-mono text-muted-foreground">#{problem.id}</span>{" "}
   389→                  {problem.title}
   390→                </h3>
   391→                <p className="mt-0.5 text-[11px] text-muted-foreground">
   392→                  Problem #{problem.id} · {problem.topic} · Acceptance{" "}
   393→                  <span className="text-emerald-600 dark:text-emerald-300">{problem.acceptance}%</span>
   394→                </p>
   395→              </div>
   396→              <Badge className={difficultyBadge}>{problem.difficulty}</Badge>
   397→            </div>
   398→
   399→            <div className="mt-4 space-y-3 text-sm text-foreground/85">
   400→              <p>{problem.description}</p>
   401→            </div>
   402→
   403→            <div className="mt-5">
   404→              <p className="text-xs uppercase tracking-wider text-muted-foreground">Examples</p>
   405→              <div className="mt-2 space-y-2">
   406→                {problem.examples.map((e, i) => (
   407→                  <div
   408→                    key={i}
   409→                    className="rounded-md border border-black/10 bg-zinc-900 p-3 font-mono text-xs dark:border-white/10"
   410→                  >
   411→                    <p className="text-emerald-300">
   412→                      Input: <span className="text-white/80">{e.input}</span>
   413→                    </p>
   414→                    <p className="mt-1 text-violet-300">
   415→                      Output: <span className="text-white/80">{e.output}</span>
   416→                    </p>
   417→                    {e.explanation && <p className="mt-1 text-zinc-400"># {e.explanation}</p>}
   418→                  </div>
   419→                ))}
   420→              </div>
   421→            </div>
   422→
   423→            <div className="mt-5">
   424→              <p className="text-xs uppercase tracking-wider text-muted-foreground">Constraints</p>
   425→              <ul className="mt-2 space-y-1 font-mono text-xs text-foreground/70">
   426→                {problem.constraints.map((c, i) => (
   427→                  <li key={i}>{c}</li>
   428→                ))}
   429→              </ul>
   430→            </div>
   431→
   432→            <div className="mt-auto pt-5">
   433→              <div className="flex items-center gap-2 text-[11px] text-muted-foreground">
   434→                <Clock className="h-3.5 w-3.5" />
   435→                Avg solve time: <span className="text-foreground">4m 12s</span> · Acceptance
   436→                rate:{" "}
   437→                <span className="text-emerald-600 dark:text-emerald-300">{problem.acceptance}%</span>
   438→              </div>
   439→            </div>
   440→          </div>
   441→
   442→          {/* RIGHT PANE: editor */}
   443→          <div className="flex h-full flex-col overflow-hidden rounded-lg border border-black/10 bg-card shadow-sm dark:border-white/10">
   444→            {/* Editor chrome */}
   445→            <div className="flex items-center justify-between border-b border-black/10 px-4 py-2 dark:border-white/5">
   446→              <span className="font-mono text-xs text-muted-foreground">solution.{fileExt}</span>
   447→              <div className="flex items-center gap-2">
   448→                <Select value={language} onValueChange={onLangChange}>
   449→                  <SelectTrigger size="sm" className="w-32 rounded-md border-black/10 bg-black/[0.02] dark:border-white/10 dark:bg-white/5">

   460→                {autoSaved && (

   478→                </div>
   479→                <textarea
   480→                  value={code}
   481→                  onChange={(e) => setCode(e.target.value)}
   482→                  spellCheck={false}
   483→                  className="flex-1 resize-none bg-transparent p-3 font-mono text-xs leading-6 text-emerald-100 outline-none"
   484→                  style={{ minHeight: 280, tabSize: 4 }}
   485→                />
   486→              </div>

   489→            {/* Custom input + actions */}
   490→            <div className="border-t border-black/10 p-4 dark:border-white/5">

   494→                onChange={(e) => setCustomInput(e.target.value)}

   496→              />
   497→              <div className="mt-3 flex flex-wrap items-center gap-2">
   498→                <Button
   499→                  variant="outline"
   500→                  size="sm"

   502→                  disabled={running}

   505→                  {running ? (
   506→                    <Clock className="mr-2 h-4 w-4 animate-spin" />
   507→                  ) : (
   508→                    <Play className="mr-2 h-4 w-4" />
   509→                  )}
   510→                  Run
   511→                </Button>
   512→                <Button
   513→                  size="sm"
   514→                  onClick={submit}
   515→                  disabled={running}
   516→                  className="rounded-full bg-violet-500 text-white hover:bg-violet-600"
   517→                >
   518→                  {running ? (
   519→                    <Clock className="mr-2 h-4 w-4 animate-spin" />
   520→                  ) : (
   521→                    <CheckCircle2 className="mr-2 h-4 w-4" />
   522→                  )}
   523→                  Submit
   524→                </Button>
   525→                <Button
   526→                  size="sm"
   527→                  variant="ghost"
   528→                  className="rounded-full text-muted-foreground"
   529→                  onClick={() => navigator.clipboard?.writeText(code)}
   530→                >
   531→                  <Copy className="h-3.5 w-3.5" /> Copy

   533→              </div>
   534→            </div>
   535→
   536→            {/* Test cases */}
   537→            {results && (
   538→              <div className="border-t border-black/10 p-4 dark:border-white/5">
   539→                <div className="flex items-center justify-between">
   540→                  <p className="text-xs uppercase tracking-wider text-muted-foreground">

   543→                  {submitted && (

   547→                  )}
   548→                </div>
   549→                <ul className="mt-3 space-y-1.5">
   550→                  {results.map((r) => {
