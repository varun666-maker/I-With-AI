   315→function SkillsTab() {
   316→  const [skills, setSkills] = useState<string[]>(["Python", "FastAPI", "React", "SQL", "PostgreSQL", "Redis"]);
   317→  const [input, setInput] = useState("");
   318→
   319→  const suggestions = useMemo(
   320→    () =>
   321→      SKILL_SUGGESTIONS.filter(
   322→        (s) =>
   323→          s.toLowerCase().includes(input.toLowerCase()) &&
   324→          !skills.some((x) => x.toLowerCase() === s.toLowerCase()),
   325→      ).slice(0, 6),
   326→    [input, skills],
   327→  );
   328→
   329→  const add = (s: string) => {
   330→    if (!s.trim()) return;
   331→    if (!skills.includes(s)) setSkills([...skills, s]);
   332→    setInput("");
   333→  };
   334→  const remove = (s: string) => setSkills(skills.filter((x) => x !== s));
   335→
   336→  return (
   337→    <TabShell
   338→      title="Skills"
   339→      desc="Tag-input with autocomplete. Each skill you add directly influences the AI's question selection in technical interviews."
   340→    >
   341→      <div>
   342→        <Label className="text-xs text-muted-foreground">Your Skills</Label>
   343→        <div className="mt-1 flex flex-wrap items-center gap-1.5 rounded-lg border border-black/10 bg-black/[0.02] p-2 dark:border-white/10 dark:bg-white/5">
   344→          {skills.map((s) => (
   345→            <Badge key={s} className="border-violet-400/30 bg-violet-500/15 text-violet-700 dark:text-violet-200">
   346→              {s}
   347→              <button onClick={() => remove(s)} className="ml-1" aria-label={`Remove ${s}`}>
   348→                <X className="h-3 w-3" />
   349→              </button>
   350→            </Badge>
   351→          ))}
   352→          <input
   353→            value={input}
   354→            onChange={(e) => setInput(e.target.value)}
   355→            onKeyDown={(e) => {
   356→              if (e.key === "Enter") { e.preventDefault(); add(input); }
   357→            }}
   358→            placeholder="Type a skill and press Enter…"
   359→            className="min-w-[140px] flex-1 bg-transparent px-1 text-sm outline-none placeholder:text-muted-foreground"
   360→          />
   361→        </div>
   362→
   363→        {suggestions.length > 0 && (
   364→          <div className="mt-2 flex flex-wrap gap-1.5">
   365→            <span className="text-[11px] text-muted-foreground">Suggestions:</span>
   366→            {suggestions.map((s) => (
   367→              <button
   368→                key={s}
   369→                onClick={() => add(s)}
   370→                className="rounded-full border border-black/10 bg-black/[0.02] px-2.5 py-1 text-[11px] text-muted-foreground hover:border-violet-400/30 hover:text-violet-700 dark:border-white/10 dark:bg-white/5 dark:hover:text-violet-300"
   371→              >
   372→                + {s}
   373→              </button>
   374→            ))}
   375→          </div>
   376→        )}
   377→      </div>
   378→
   379→      <div className="mt-6 grid grid-cols-2 gap-3 sm:grid-cols-4">
   380→        <SkillStat label="Total skills" value={String(skills.length)} tone="text-violet-500 dark:text-violet-300" />
   381→        <SkillStat label="Backend" value="4" tone="text-emerald-600 dark:text-emerald-300" />
   382→        <SkillStat label="Frontend" value="1" tone="text-amber-600 dark:text-amber-300" />
   383→        <SkillStat label="Infra" value="1" tone="text-sky-600 dark:text-sky-300" />
   384→      </div>
   385→
   386→      <div className="mt-6 rounded-md border border-emerald-400/20 bg-emerald-500/[0.04] p-4 text-xs text-emerald-700 dark:text-emerald-200">
   387→        <Badge variant="outline" className="mb-1 border-emerald-400/30 bg-emerald-500/10 text-[10px] uppercase tracking-widest text-emerald-700 dark:text-emerald-200">
   388→          Profile completeness
   389→        </Badge>
   390→        <p>
   391→          Your profile is <strong>82% complete</strong>. Adding 2 more backend skills (Docker, CI/CD) would push you to 100%, unlocking more granular interview question sets.
   392→        </p>
   393→      </div>
   394→    </TabShell>
   395→  );

   463→              <SelectContent>
   464→                {EXP_OPTIONS.map((o) => (
   465→                  <SelectItem key={o} value={o}>{o}</SelectItem>
   466→                ))}
   467→              </SelectContent>
   468→            </Select>
   469→          </div>
   470→          <div>
   471→            <Label className="text-xs text-muted-foreground">Expected Salary (CTC)</Label>

   475→              placeholder="e.g. 18-25 LPA"
   476→              className="mt-1 rounded-lg bg-black/[0.02] border-black/10 dark:bg-white/5 dark:border-white/10"
   477→            />
   478→          </div>
   479→        </div>
   480→
   481→        {/* Locations */}
   482→        <div>
   483→          <Label className="text-xs text-muted-foreground">Preferred Locations</Label>
   484→          <div className="mt-1 flex flex-wrap items-center gap-1.5 rounded-lg border border-black/10 bg-black/[0.02] p-2 dark:border-white/10 dark:bg-white/5">

   487→                <MapPin className="mr-1 h-3 w-3" />

   516→            </div>

   518→        </div>
   519→
   520→        {/* Notice + Remote + Shift */}

   526→                <SelectValue />
   527→              </SelectTrigger>
   528→              <SelectContent>
   529→                <SelectItem value="0">Immediate</SelectItem>
   530→                <SelectItem value="15">15 days</SelectItem>
   531→                <SelectItem value="30">30 days</SelectItem>
   532→                <SelectItem value="60">60 days</SelectItem>
   533→                <SelectItem value="90">90 days</SelectItem>
   534→              </SelectContent>
   535→            </Select>
   536→          </div>
   537→          <div>

   539→            <Select defaultValue="hybrid">
   540→              <SelectTrigger className="mt-1 w-full rounded-lg bg-black/[0.02] border-black/10 dark:bg-white/5 dark:border-white/10">
   541→                <SelectValue />
   542→              </SelectTrigger>
   543→              <SelectContent>
   544→                <SelectItem value="remote">Remote only</SelectItem>

   549→          </div>
   550→          <div>
   551→            <Label className="text-xs text-muted-foreground">Shift Preference</Label>
   552→            <Select defaultValue="day">
   553→              <SelectTrigger className="mt-1 w-full rounded-lg bg-black/[0.02] border-black/10 dark:bg-white/5 dark:border-white/10">
   554→                <SelectValue />
   555→              </SelectTrigger>
   556→              <SelectContent>
   557→                <SelectItem value="day">Day shift</SelectItem>
   558→                <SelectItem value="rotational">Rotational</SelectItem>
   559→                <SelectItem value="night">Night shift</SelectItem>
   560→              </SelectContent>
   561→            </Select>
   562→          </div>
   563→        </div>
   564→
   565→        <div className="rounded-md border border-violet-400/20 bg-violet-500/[0.04] p-4 text-xs text-violet-700 dark:text-violet-200">
   566→          <Info className="mr-1 inline h-3.5 w-3.5" />
   567→          Saved preferences are visible to recruiters in Hiring Mode. You can opt out anytime under <em>Settings → Visibility</em>.
   568→        </div>
   569→      </div>
   570→    </TabShell>
   571→  );
   572→}
