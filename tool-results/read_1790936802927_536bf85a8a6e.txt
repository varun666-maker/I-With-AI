   112→   112→/**
   113→   113→ * Fisher-Yates shuffle. Returns a new array (does not mutate the input).
   114→   114→ * Used to randomize the order of problems each candidate sees, so the
   115→   115→ * first problem shown when entering the coding test is random.
   116→   116→ */
   117→   117→function shuffle<T>(arr: readonly T[]): T[] {
   118→   118→  const out = arr.slice();
   119→   119→  for (let i = out.length - 1; i > 0; i--) {
   120→   120→    const j = Math.floor(Math.random() * (i + 1));
   121→   121→    [out[i], out[j]] = [out[j], out[i]];
   122→   122→  }
   123→   123→  return out;
   124→   124→}
