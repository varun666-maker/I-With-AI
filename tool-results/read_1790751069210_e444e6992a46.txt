   745→  tag,
   746→  tone,
   747→  icon,
   748→}: {
   749→  tag: string;
   750→  tone: "violet" | "emerald" | "amber" | "rose" | "sky" | "slate";
   751→  icon?: React.ReactNode;

   753→  const tones: Record<string, string> = {

   755→    emerald: "border-emerald-400/30 bg-emerald-500/10 text-emerald-700 dark:text-emerald-300",

   757→    rose: "border-rose-400/30 bg-rose-500/10 text-rose-700 dark:text-rose-300",
   758→    sky: "border-sky-400/30 bg-sky-500/10 text-sky-700 dark:text-sky-300",
   759→    slate: "border-black/15 bg-muted/40 text-muted-foreground dark:border-white/15",

   764→        "inline-flex items-center gap-1 rounded-full border px-2.5 py-0.5 text-xs font-medium",

   770→    </span>
   771→  );
